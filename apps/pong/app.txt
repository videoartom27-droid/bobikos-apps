REM ---- PONG ----

LABEL start
LET PX = 50
LET BX = 50
LET X = 64
LET Y = 32
LET DX = 2
LET DY = 2
LET PS = 0
LET BS = 0

LABEL loop
CLS

TXT 20,0,"BOT"
NUM 40,0,BS
TXT 75,0,"YOU"
NUM 100,0,PS

RECT PX,58,28,4,2
RECT BX,2,28,4,2
CIRC X,Y,2,2

IF BTN(LEFT)==1 GOTO left
IF BTN(RIGHT)==1 GOTO right
GOTO inputDone

LABEL left
LET PX = PX-3
GOTO inputDone

LABEL right
LET PX = PX+3

LABEL inputDone
IF PX<0 GOTO playerMin
IF PX>100 GOTO playerMax
GOTO bot

LABEL playerMin
LET PX = 0
GOTO bot

LABEL playerMax
LET PX = 100

LABEL bot
IF X<BX+10 GOTO botLeft
IF X>BX+18 GOTO botRight
GOTO botDone

LABEL botLeft
LET BX = BX-1
GOTO botDone

LABEL botRight
LET BX = BX+1

LABEL botDone
IF BX<0 GOTO botMin
IF BX>100 GOTO botMax
GOTO move

LABEL botMin
LET BX = 0
GOTO move

LABEL botMax
LET BX = 100

LABEL move
LET X = X+DX
LET Y = Y+DY

IF X<3 GOTO hitLeft
IF X>124 GOTO hitRight
GOTO checkPlayer

LABEL hitLeft
LET X = 3
LET DX = 2
GOTO checkPlayer

LABEL hitRight
LET X = 124
LET DX = -2

LABEL checkPlayer
IF Y<55 GOTO checkBot
IF Y>58 GOTO playerMiss
IF X<PX GOTO checkBot
IF X>PX+28 GOTO checkBot
LET DY = -2
BEEP 1200,20
GOTO draw

LABEL playerMiss
LET BS = BS+1
BEEP 200,100
LET X = 64
LET Y = 32
LET DX = -2
LET DY = -2
GOTO draw

LABEL checkBot
IF Y>7 GOTO draw
IF Y<5 GOTO botMiss
IF X<BX GOTO botMiss
IF X>BX+28 GOTO botMiss
LET DY = 2
BEEP 1200,20
GOTO draw

LABEL botMiss
LET PS = PS+1
BEEP 2000,100
LET X = 64
LET Y = 32
LET DX = 2
LET DY = 2

LABEL draw
SHOW
WAIT 40
GOTO loop
