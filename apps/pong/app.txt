REM ---- PONG: игрок (слева, UP/DOWN) против простого AI (справа) ----
LABEL start
LET A = 26
LET B = 26
LET S = 0
LET T = 0
GOSUB resetball

LABEL loop
CLS

REM --- движение ракетки игрока ---
IF BTN(UP)==1 GOTO pup
IF BTN(DOWN)==1 GOTO pdown
GOTO pclamp
LABEL pup
LET A = A-3
GOTO pclamp
LABEL pdown
LET A = A+3
GOTO pclamp
LABEL pclamp
IF A<0 GOTO pmin
IF A>52 GOTO pmax
GOTO afterp
LABEL pmin
LET A = 0
GOTO afterp
LABEL pmax
LET A = 52
GOTO afterp
LABEL afterp

REM --- простой AI для правой ракетки ---
IF D<B+6 GOTO aup
IF D>B+6 GOTO adown
GOTO aclamp
LABEL aup
LET B = B-2
GOTO aclamp
LABEL adown
LET B = B+2
GOTO aclamp
LABEL aclamp
IF B<0 GOTO bmin
IF B>52 GOTO bmax
GOTO aftera
LABEL bmin
LET B = 0
GOTO aftera
LABEL bmax
LET B = 52
GOTO aftera
LABEL aftera

REM --- движение мяча ---
LET C = C+E
LET D = D+F

IF D<=0 GOTO topbounce
IF D>=62 GOTO botbounce
GOTO afterwall
LABEL topbounce
LET F = 1
LET D = 0
GOTO afterwall
LABEL botbounce
LET F = -1
LET D = 62
GOTO afterwall
LABEL afterwall

REM --- отскок от ракеток ---
IF C<=6 AND D>=A AND D<=A+12 AND E==-1 GOTO hitp
GOTO checka
LABEL hitp
LET E = 1
BEEP 1800,15
GOTO afterhit
LABEL checka
IF C>=120 AND D>=B AND D<=B+12 AND E==1 GOTO hita
GOTO afterhit
LABEL hita
LET E = -1
BEEP 1800,15
LABEL afterhit

REM --- голы ---
IF C<0 GOTO aiscore
IF C>127 GOTO pscore
GOTO draw
LABEL aiscore
LET T = T+1
BEEP 200,150
GOSUB resetball
GOTO draw
LABEL pscore
LET S = S+1
BEEP 2500,80
GOSUB resetball

LABEL draw
RECT 4,A,2,12,2
RECT 122,B,2,12,2
CIRC C,D,2,2
LINE 64,0,64,63,1
NUM 40,0,S
NUM 80,0,T
SHOW
WAIT 30
GOTO loop

LABEL resetball
LET C = 64
LET D = 32
LET E = 1
LET F = 1
RETURN
