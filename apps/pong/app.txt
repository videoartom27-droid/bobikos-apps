REM ---- PONG ----

LABEL start
LET PY = 31
LET BY = 31
LET X = 64
LET Y = 32
LET DX = 3
LET DY = 1
LET PS = 0
LET BS = 0

LABEL loop
CLS

TXT 48,0,"YOU"
NUM 64,0,PS
TXT 82,0,"BOT"
NUM 106,0,BS

LINE 64,13,64,17,1
LINE 64,20,64,24,1
LINE 64,27,64,31,1
LINE 64,34,64,38,1
LINE 64,41,64,45,1
LINE 64,48,64,52,1
LINE 64,55,64,59,1

RECT 2,PY,4,14,2
RECT 122,BY,4,14,2
CIRC X,Y,2,2

IF BTN(UP)==1 GOTO playerUp
IF BTN(DOWN)==1 GOTO playerDown
GOTO inputDone

LABEL playerUp
LET PY = PY-4
GOTO inputDone

LABEL playerDown
LET PY = PY+4

LABEL inputDone
IF PY<14 GOTO playerMin
IF PY>46 GOTO playerMax
GOTO bot

LABEL playerMin
LET PY = 14
GOTO bot

LABEL playerMax
LET PY = 46

LABEL bot
IF Y<BY+7 GOTO botUp
IF Y>BY+7 GOTO botDown
GOTO botDone

LABEL botUp
LET BY = BY-4
GOTO botDone

LABEL botDown
LET BY = BY+4

LABEL botDone
IF BY<14 GOTO botMin
IF BY>46 GOTO botMax
GOTO move

LABEL botMin
LET BY = 14
GOTO move

LABEL botMax
LET BY = 46

LABEL move
LET X = X+DX
LET Y = Y+DY

IF Y<16 GOTO topWall
IF Y>61 GOTO bottomWall
GOTO paddleCheck

LABEL topWall
LET Y = 16
LET DY = 0-DY
GOTO paddleCheck

LABEL bottomWall
LET Y = 61
LET DY = 0-DY

LABEL paddleCheck
IF DX<0 GOTO leftCheck
GOTO rightCheck

LABEL leftCheck
IF X>9 GOTO draw
IF Y<PY GOTO playerMiss
IF Y>PY+14 GOTO playerMiss

LET X = 7
LET DX = 3

IF Y<PY+3 GOTO playerHardUp
IF Y<PY+6 GOTO playerUp
IF Y<PY+10 GOTO playerDown
IF Y<PY+13 GOTO playerHardDown
GOTO playerCenter

LABEL playerHardUp
LET DY = -3
GOTO draw

LABEL playerUp
LET DY = -2
GOTO draw

LABEL playerCenter
IF DY<0 GOTO centerUp
LET DY = 1
GOTO draw

LABEL centerUp
LET DY = -1
GOTO draw

LABEL playerDown
LET DY = 2
GOTO draw

LABEL playerHardDown
LET DY = 3
GOTO draw

LABEL rightCheck
IF X<116 GOTO draw
IF Y<BY GOTO botMiss
IF Y>BY+14 GOTO botMiss

LET X = 118
LET DX = -3

IF Y<BY+3 GOTO botHardUp
IF Y<BY+6 GOTO botUp
IF Y<BY+10 GOTO botDown
IF Y<BY+13 GOTO botHardDown
GOTO botCenter

LABEL botHardUp
LET DY = -3
GOTO draw

LABEL botUp
LET DY = -2
GOTO draw

LABEL botCenter
IF DY<0 GOTO botCenterUp
LET DY = 1
GOTO draw

LABEL botCenterUp
LET DY = -1
GOTO draw

LABEL botDown
LET DY = 2
GOTO draw

LABEL botHardDown
LET DY = 3
GOTO draw

LABEL playerMiss
LET BS = BS+1
BEEP 200,80
LET X = 64
LET Y = RND(20)+22
LET DX = 3
LET DY = 1
GOTO draw

LABEL botMiss
LET PS = PS+1
BEEP 2000,80
LET X = 64
LET Y = RND(20)+22
LET DX = -3
LET DY = -1

LABEL draw
SHOW
WAIT 5
GOTO loop

LABEL loop

NUM 104,0,BS

LINE 64,13,64,17,1
LINE 64,20,64,24,1

RECT 2,PY,4,14,2
RECT 122,BY,4,14,2
CIRC X,Y,2,2

IF BTN(UP)==1 GOTO playerUp
IF BTN(DOWN)==1 GOTO playerDown
GOTO inputDone

LABEL playerUp
LET PY = PY-3
GOTO inputDone

LABEL playerDown

LABEL inputDone
IF PY<14 GOTO playerMin
IF PY>46 GOTO playerMax
GOTO bot

LABEL playerMin
LET PY = 14
GOTO bot

LABEL playerMax
LET PY = 46

LABEL bot
IF X<70 GOTO botWait
IF Y<BY+3 GOTO botUp
IF Y>BY+11 GOTO botDown
GOTO botDone

LABEL botUp
LET BY = BY-3
GOTO botDone

LABEL botDown
LET BY = BY+3
GOTO botDone

LABEL botWait
IF BY<32 GOTO botCenterUp
IF BY>32 GOTO botCenterDown
GOTO botDone

LABEL botCenterUp
LET BY = BY+1
GOTO botDone

LABEL botCenterDown
LET BY = BY-1

LABEL botDone
IF BY<14 GOTO botMin
IF BY>46 GOTO botMax
GOTO move

LABEL botMin
LET BY = 14
GOTO move

LABEL botMax
LET BY = 46

LABEL move
LET X = X+DX
LET Y = Y+DY

IF Y<15 GOTO topWall
IF Y>61 GOTO bottomWall
GOTO collision

LABEL topWall
LET Y = 15
LET DY = 0-DY
GOTO collision

LABEL bottomWall
LET Y = 61
LET DY = 0-DY

LABEL collision
IF DX<0 GOTO leftCollision
GOTO rightCollision

LABEL leftCollision
IF X>8 GOTO draw
IF X<4 GOTO playerMiss
IF Y<PY GOTO playerMiss
IF Y>PY+14 GOTO playerMiss

LET X = 8
LET DX = SPEED

IF Y<PY+3 GOTO playerHardUp
IF Y<PY+6 GOTO playerUpHit
IF Y<PY+10 GOTO playerCenterHit
IF Y<PY+13 GOTO playerDownHit
GOTO playerHardDown

LABEL playerHardUp
LET DY = -3
GOTO hit

LABEL playerUpHit
LET DY = -2
GOTO hit

LABEL playerCenterHit
LET DY = 0
GOTO hit

LABEL playerDownHit
LET DY = 2
GOTO hit

LABEL playerHardDown
LET DY = 3

LABEL hit
LET SPEED = SPEED+1
IF SPEED>5 GOTO speedMax
GOTO draw

LABEL speedMax
LET SPEED = 5
GOTO draw

LABEL rightCollision
IF X<118 GOTO draw
IF X>122 GOTO botMiss
IF Y<BY GOTO botMiss
IF Y>BY+14 GOTO botMiss

LET X = 118
LET DX = 0-SPEED

IF Y<BY+3 GOTO botHardUp
IF Y<BY+6 GOTO botUpHit
IF Y<BY+10 GOTO botCenterHit
IF Y<BY+13 GOTO botDownHit
GOTO botHardDown

LABEL botHardUp
LET DY = -3
GOTO hitBot

LABEL botUpHit
LET DY = -2
GOTO hitBot

LABEL botCenterHit
LET DY = 0
GOTO hitBot

LABEL botDownHit
LET DY = 2
GOTO hitBot

LABEL botHardDown
LET DY = 3

LABEL hitBot
LET SPEED = SPEED+1
IF SPEED>5 GOTO speedMaxBot
GOTO draw

LABEL speedMaxBot
LET SPEED = 5
GOTO draw

LABEL playerMiss
LET BS = BS+1
BEEP 200,80
LET X = 64
LET Y = 32
LET DX = 3
LET DY = RND(4)-2
LET SPEED = 3
GOTO draw

LABEL botMiss
LET PS = PS+1

LET DX = -3
