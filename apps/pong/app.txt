REM ---- PONG ----
LABEL start
LET PY = 25
LET BY = 25
LET X = 64
LET Y = 32
LET DX = 4
LET DY = 2
LET S = 0

LABEL loop
CLS

LINE 64,14,64,17,1
LINE 64,20,64,23,1
LINE 64,26,64,29,1
LINE 64,32,64,35,1
LINE 64,38,64,41,1
LINE 64,44,64,47,1
LINE 64,50,64,53,1

RECT 2,PY,4,14,2
RECT 122,BY,4,14,2
CIRC X,Y,2,2

IF BTN(UP)==1 GOTO playerUp
IF BTN(DOWN)==1 GOTO playerDown
GOTO bot

LABEL playerUp
LET PY = PY-4
IF PY<14 GOTO playerMin
GOTO bot

LABEL playerMin
LET PY = 14
GOTO bot

LABEL playerDown
LET PY = PY+4
IF PY>46 GOTO playerMax
GOTO bot

LABEL playerMax
LET PY = 46

LABEL bot
IF DX<0 GOTO move
IF Y<BY+5 GOTO botUp
IF Y>BY+9 GOTO botDown
GOTO move

LABEL botUp
LET BY = BY-2
IF BY<14 GOTO botMin
GOTO move

LABEL botMin
LET BY = 14
GOTO move

LABEL botDown
LET BY = BY+2
IF BY>46 GOTO botMax
GOTO move

LABEL botMax
LET BY = 46

LABEL move
LET X = X+DX
LET Y = Y+DY

IF Y<=16 GOTO topWall
IF Y>=61 GOTO bottomWall
GOTO paddleCheck

LABEL topWall
LET Y = 16
LET DY = 0-DY
GOTO paddleCheck

LABEL bottomWall
LET Y = 61
LET DY = 0-DY

LABEL paddleCheck
IF DX<0 GOTO leftCheck
GOTO rightCheck

LABEL leftCheck
IF X>7 GOTO draw
IF Y<PY-2 GOTO resetPlayer
IF Y>PY+16 GOTO resetPlayer

LET X = 8
LET DX = 4
LET O = Y-PY
LET S = 1-S

IF O<3 GOTO leftHardUp
IF O<6 GOTO leftUp
IF O<9 GOTO leftCenter
IF O<12 GOTO leftDown
GOTO leftHardDown

LABEL leftHardUp
LET DY = -3
GOTO draw

LABEL leftUp
LET DY = -2
GOTO draw

LABEL leftCenter
IF S==0 GOTO leftCenterUp
LET DY = 1
GOTO draw

LABEL leftCenterUp
LET DY = -1
GOTO draw

LABEL leftDown
LET DY = 2
GOTO draw

LABEL leftHardDown
LET DY = 3
GOTO draw

LABEL rightCheck
IF X<120 GOTO draw
IF Y<BY-2 GOTO resetBot
IF Y>BY+16 GOTO resetBot

LET X = 118
LET DX = -4
LET O = Y-BY
LET S = 1-S

IF O<3 GOTO rightHardUp
IF O<6 GOTO rightUp
IF O<9 GOTO rightCenter
IF O<12 GOTO rightDown
GOTO rightHardDown

LABEL rightHardUp
LET DY = -3
GOTO draw

LABEL rightUp
LET DY = -2
GOTO draw

LABEL rightCenter
IF S==0 GOTO rightCenterUp
LET DY = 1
GOTO draw

LABEL rightCenterUp
LET DY = -1
GOTO draw

LABEL rightDown
LET DY = 2
GOTO draw

LABEL rightHardDown
LET DY = 3
GOTO draw

LABEL resetPlayer
LET X = 64
LET Y = 32
LET DX = 4
LET DY = 2
GOTO draw

LABEL resetBot
LET X = 64
LET Y = 32
LET DX = -4
LET DY = -2
GOTO draw

LABEL draw
SHOW
WAIT 5
GOTO loop
