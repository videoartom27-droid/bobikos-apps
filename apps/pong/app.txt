REM ---- PONG: игрок против бота ----

LABEL start
LET PX = 50
LET BX = 50
LET X = 64
LET Y = 32
LET DX = 2
LET DY = 2
LET PS = 0
LET BS = 0

LABEL loop
CLS

RECT PX,60,28,4,2
RECT BX,2,28,4,2
CIRC X,Y,2,2

NUM 0,0,BS
NUM 110,0,PS

IF BTN(LEFT)==1 GOTO playerLeft
IF BTN(RIGHT)==1 GOTO playerRight
GOTO bot

LABEL playerLeft
LET PX = PX-4
GOTO bot

LABEL playerRight
LET PX = PX+4
GOTO bot

LABEL bot
IF X<BX GOTO botLeft
IF X>BX+28 GOTO botRight
GOTO moveBall

LABEL botLeft
LET BX = BX-2
GOTO moveBall

LABEL botRight
LET BX = BX+2
GOTO moveBall

LABEL moveBall
IF PX<0 GOTO playerMin
IF PX>100 GOTO playerMax
IF BX<0 GOTO botMin
IF BX>100 GOTO botMax
GOTO ball

LABEL playerMin
LET PX = 0
GOTO ball

LABEL playerMax
LET PX = 100
GOTO ball

LABEL botMin
LET BX = 0
GOTO ball

LABEL botMax
LET BX = 100
GOTO ball

LABEL ball
LET X = X+DX
LET Y = Y+DY

IF X<2 GOTO wallLeft
IF X>125 GOTO wallRight
GOTO paddleCheck

LABEL wallLeft
LET X = 2
LET DX = 2
GOTO paddleCheck

LABEL wallRight
LET X = 125
LET DX = -2
GOTO paddleCheck

LABEL paddleCheck
IF Y>56 GOTO playerCheck
IF Y<8 GOTO botCheck
GOTO draw

LABEL playerCheck
IF X<PX GOTO draw
IF X>PX+28 GOTO draw
LET DY = -2
GOTO draw

LABEL botCheck
IF X<BX GOTO draw
IF X>BX+28 GOTO draw
LET DY = 2
GOTO draw

LABEL draw
SHOW
WAIT 40
IF Y>66 GOTO botScore
IF Y<-2 GOTO playerScore
GOTO loop

LABEL playerScore
LET PS = PS+1
BEEP 2000,100
LET X = 64
LET Y = 32
LET DX = 2
LET DY = 2
GOTO loop

LABEL botScore
LET BS = BS+1
BEEP 200,100
LET X = 64
LET Y = 32
LET DX = -2
LET DY = -2
GOTO loop
